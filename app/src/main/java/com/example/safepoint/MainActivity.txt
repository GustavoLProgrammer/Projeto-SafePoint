// ================= PREFS HELPER =================
object PrefsHelper {
    fun getPrefs(context: Context): SharedPreferences =
        context.getSharedPreferences("safepoint_prefs", Context.MODE_PRIVATE)
    fun getUserId(context: Context): String? = getPrefs(context).getString("user_id", null)
    fun setUserId(context: Context, id: String) { getPrefs(context).edit().putString("user_id", id).apply() }
    fun clearUserId(context: Context) { getPrefs(context).edit().remove("user_id").apply() }
    fun isTermosAceitos(context: Context): Boolean = getPrefs(context).getBoolean("termos_aceitos", false)
    fun setTermosAceitos(context: Context, aceito: Boolean) { getPrefs(context).edit().putBoolean("termos_aceitos", aceito).apply() }
    fun isLocalizacaoAceita(context: Context): Boolean = getPrefs(context).getBoolean("loc_aceita", false)
    fun setLocalizacaoAceita(context: Context, aceito: Boolean) { getPrefs(context).edit().putBoolean("loc_aceita", aceito).apply() }
}

// ================= MAIN ACTIVITY =================
class MainActivity : ComponentActivity() {
    private lateinit var auth: FirebaseAuth
    private lateinit var db: FirebaseFirestore

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        auth = FirebaseAuth.getInstance()
        db = FirebaseFirestore.getInstance()

        // OSMDroid config profissional
        Configuration.getInstance().userAgentValue = packageName
        Configuration.getInstance().load(this, getSharedPreferences("osmdroid_prefs", Context.MODE_PRIVATE))
        Configuration.getInstance().osmdroidTileCache.delete()

        setContent {
            MaterialTheme {
                val navController = rememberNavController()
                val context = LocalContext.current
                val termosAceitos = PrefsHelper.isTermosAceitos(context)
                val locAceita = PrefsHelper.isLocalizacaoAceita(context)
                val currentUser = auth.currentUser

                val startDest = when {
                    !termosAceitos -> Screen.Termos.route
                    !locAceita -> Screen.PermissaoLocalizacao.route
                    currentUser == null -> Screen.Login.route
                    else -> Screen.Home.route
                }

                NavHost(navController = navController, startDestination = startDest) {
                    composable(Screen.Termos.route) {
                        TelaTermosUso(
                            onAceitar = {
                                PrefsHelper.setTermosAceitos(context, true)
                                navController.navigate(Screen.PermissaoLocalizacao.route) {
                                    popUpTo(Screen.Termos.route) { inclusive = true }
                                }
                            }
                        )
                    }
                    composable(Screen.PermissaoLocalizacao.route) {
                        TelaPermissaoLocalizacao(
                            onAceitar = {
                                PrefsHelper.setLocalizacaoAceita(context, true)
                                if (auth.currentUser != null) {
                                    navController.navigate(Screen.Home.route) { popUpTo(0) { inclusive = true } }
                                } else {
                                    navController.navigate(Screen.Login.route) { popUpTo(0) { inclusive = true } }
                                }
                            },
                            onRecusar = {
                                Toast.makeText(context, "É necessário permitir localização para usar o mapa", Toast.LENGTH_LONG).show()
                            }
                        )
                    }
                    composable(Screen.Login.route) { TelaLogin(navController, auth) }
                    composable(Screen.Cadastro.route) { TelaCadastro(navController, auth) }
                    composable(Screen.RecuperarSenha.route) { TelaRecuperarSenha(navController, auth) }
                    composable(Screen.Home.route) { TelaHome(navController) }
                    composable(Screen.Mapa.route) { MapaScreen(onVoltar = { navController.popBackStack() }) }
                    composable(Screen.Historico.route) { TelaHistorico(navController) }
                    composable(Screen.Perfil.route) { TelaPerfil(navController, auth, db) }
                    composable(Screen.SOS.route) { TelaSOS(onVoltar = { navController.popBackStack() }) }
                    composable(Screen.Ajuda.route) { TelaAjuda(onVoltar = { navController.popBackStack() }) }
                }
            }
        }
    }
}

// ================= TELA PERMISSÃO LOCALIZAÇÃO - NOVA =================
@Composable
fun TelaPermissaoLocalizacao(onAceitar: () -> Unit, onRecusar: () -> Unit) {
    val context = LocalContext.current
    var temPermissao by remember { mutableStateOf(ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED) }
    val launcher = rememberLauncherForActivityResult(ActivityResultContracts.RequestMultiplePermissions()) { perms ->
        temPermissao = perms[Manifest.permission.ACCESS_FINE_LOCATION] == true || perms[Manifest.permission.ACCESS_COARSE_LOCATION] == true
        if (temPermissao) onAceitar() else onRecusar()
    }

    LaunchedEffect(temPermissao) { if (temPermissao) onAceitar() }

    Box(modifier = Modifier.fillMaxSize().background(FundoGradiente), contentAlignment = Alignment.Center) {
        Card(shape = RoundedCornerShape(24.dp), modifier = Modifier.fillMaxWidth().padding(24.dp)) {
            Column(modifier = Modifier.padding(28.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                Icon(Icons.Default.LocationOn, contentDescription = null, modifier = Modifier.size(72.dp), tint = CorAzulEscuro)
                Spacer(modifier = Modifier.height(16.dp))
                Text("Permissão de Localização", fontSize = 22.sp, fontWeight = FontWeight.Bold, color = CorAzulEscuro, textAlign = TextAlign.Center)
                Spacer(modifier = Modifier.height(12.dp))
                Text(
                    "O SafePoint precisa acessar sua localização exata para mostrar rotas de evacuação, abrigos próximos e alertas reais da sua região, igual ao Google Maps.",
                    fontSize = 14.sp, color = Color.Gray, textAlign = TextAlign.Center
                )
                Spacer(modifier = Modifier.height(24.dp))
                BotaoGradiente(texto = "Permitir Localização") {
                    launcher.launch(arrayOf(Manifest.permission.ACCESS_FINE_LOCATION, Manifest.permission.ACCESS_COARSE_LOCATION))
                }
                Spacer(modifier = Modifier.height(12.dp))
                TextButton(onClick = { onRecusar() }) { Text("Agora não", color = Color.Gray) }
            }
        }
    }
}

@Composable
fun TelaLogin(navController: NavController, auth: FirebaseAuth) {
    var email by remember { mutableStateOf("") }
    var senha by remember { mutableStateOf("") }
    var senhaVisivel by remember { mutableStateOf(false) }
    var lembrarMe by remember { mutableStateOf(false) }
    val context = LocalContext.current

    Box(modifier = Modifier.fillMaxSize().background(FundoGradiente), contentAlignment = Alignment.Center) {
        Card(shape = RoundedCornerShape(32.dp), colors = CardDefaults.cardColors(containerColor = Color.White), modifier = Modifier.fillMaxWidth().padding(horizontal = 24.dp)) {
            Column(modifier = Modifier.padding(horizontal = 32.dp, vertical = 40.dp).fillMaxWidth(), horizontalAlignment = Alignment.CenterHorizontally) {
                LogoPinSafepoint()
                Spacer(modifier = Modifier.height(8.dp))
                Text(text = "Sempre te guiando a um ponto seguro", fontSize = 13.sp, color = Color.Gray)
                Spacer(modifier = Modifier.height(28.dp))
                Text(text = "Faça seu login", fontSize = 26.sp, fontWeight = FontWeight.Bold, color = CorAzulEscuro)
                Spacer(modifier = Modifier.height(28.dp))
                OutlinedTextField(value = email, onValueChange = { email = it }, placeholder = { Text("E-mail", color = Color.Gray) }, leadingIcon = { Icon(Icons.Outlined.Person, null, tint = CorAzulEscuro) }, modifier = Modifier.fillMaxWidth(), shape = RoundedCornerShape(14.dp), singleLine = true)
                Spacer(modifier = Modifier.height(16.dp))
                OutlinedTextField(value = senha, onValueChange = { senha = it }, placeholder = { Text("Senha", color = Color.Gray) }, leadingIcon = { Icon(Icons.Outlined.Lock, null, tint = CorAzulEscuro) }, trailingIcon = { IconButton(onClick = { senhaVisivel = !senhaVisivel }) { Icon(if (senhaVisivel) Icons.Default.Visibility else Icons.Default.VisibilityOff, null, tint = Color.Gray) } }, visualTransformation = if (senhaVisivel) VisualTransformation.None else PasswordVisualTransformation(), modifier = Modifier.fillMaxWidth(), shape = RoundedCornerShape(14.dp), singleLine = true)
                Spacer(modifier = Modifier.height(16.dp))
                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween, verticalAlignment = Alignment.CenterVertically) {
                    Row(verticalAlignment = Alignment.CenterVertically) { Checkbox(checked = lembrarMe, onCheckedChange = { lembrarMe = it }); Text("Lembrar-me", fontSize = 14.sp) }
                    TextButton(onClick = { navController.navigate(Screen.RecuperarSenha.route) }) { Text("Esqueci minha senha", color = CorAzulMedio, fontSize = 13.sp) }
                }
                Spacer(modifier = Modifier.height(24.dp))
                BotaoGradiente(texto = "Entrar") {
                    if (email.isNotBlank() && senha.isNotBlank()) {
                        auth.signInWithEmailAndPassword(email, senha).addOnSuccessListener {
                            PrefsHelper.setUserId(context, it.user?.uid ?: "")
                            navController.navigate(Screen.Home.route) { popUpTo(Screen.Login.route) { inclusive = true } }
                        }.addOnFailureListener { Toast.makeText(context, "Erro: ${it.localizedMessage}", Toast.LENGTH_LONG).show() }
                    } else Toast.makeText(context, "Preencha todos os campos", Toast.LENGTH_SHORT).show()
                }
                Spacer(modifier = Modifier.height(20.dp))
                Row { Text("Não tem uma conta? ", color = Color.Gray, fontSize = 13.sp); Text("Criar conta", color = CorAzulMedio, fontSize = 13.sp, fontWeight = FontWeight.Bold, modifier = Modifier.clickable { navController.navigate(Screen.Cadastro.route) }) }
            }
        }
    }
}

// ================= MAPA REAL PROFISSIONAL - CORRIGIDO =================
val NOMES_DESASTRES = mapOf("EQ" to "Terremoto", "TC" to "Ciclone", "FL" to "Enchente", "VO" to "Vulcão", "DR" to "Seca", "WF" to "Incêndio")
val NOMES_SEGURANCA = mapOf("hospital" to "Hospital", "fire_station" to "Bombeiros", "police" to "Polícia", "shelter" to "Abrigo", "pharmacy" to "Farmácia")
val CORES_SEGURANCA = mapOf("hospital" to Color(0xFFD32F2F), "fire_station" to Color(0xFFF57C00), "police" to Color(0xFF1565C0), "shelter" to Color(0xFF2E7D32), "pharmacy" to Color(0xFF7B1FA2))

@Composable
fun MapaScreen(onVoltar: () -> Unit) {
    val context = LocalContext.current
    val desastresGlobais = remember { mutableStateListOf<Ocorrencia>() }
    val pontosSeguranca = remember { mutableStateListOf<Ocorrencia>() }
    var ocorrenciaSelecionada by remember { mutableStateOf<Ocorrencia?>(null) }
    var mostrarDialog by remember { mutableStateOf(false) }
    var filtrosDesastres by remember { mutableStateOf(setOf("EQ", "TC", "FL", "VO", "DR", "WF")) }
    var filtrosSeguranca by remember { mutableStateOf(setOf<String>()) }
    var userLocation by remember { mutableStateOf<GeoPoint?>(null) }

    // Permissão
    var temPermissao by remember { mutableStateOf(ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED) }
    val permissionLauncher = rememberLauncherForActivityResult(ActivityResultContracts.RequestMultiplePermissions()) { perms ->
        temPermissao = perms[Manifest.permission.ACCESS_FINE_LOCATION] == true || perms[Manifest.permission.ACCESS_COARSE_LOCATION] == true
    }

    // MapView corrigido - usa MAPNIK oficial
    val mapView = remember {
        MapView(context).apply {
            setTileSource(TileSourceFactory.MAPNIK) // Fonte real, estável, estilo Google Maps
            setMultiTouchControls(true)
            isTilesScaledToDpi = true
            setUseDataConnection(true)
            controller.setZoom(14.0)
            controller.setCenter(GeoPoint(-23.5505, -46.6333)) // São Paulo inicial
        }
    }

    val myLocationOverlay = remember {
        MyLocationNewOverlay(GpsMyLocationProvider(context), mapView).apply {
            enableMyLocation()
            enableFollowLocation()
        }
    }

    LaunchedEffect(temPermissao) {
        if (temPermissao) {
            if (mapView.overlays.none { it is MyLocationNewOverlay }) {
                mapView.overlays.add(myLocationOverlay)
            }
            // Busca localização real
            obterLocalizacao(context) { lat, lng ->
                val ponto = GeoPoint(lat, lng)
                userLocation = ponto
                mapView.controller.animateTo(ponto)
                mapView.controller.setZoom(16.0)
                // Buscar pontos de segurança reais próximos
                kotlinx.coroutines.CoroutineScope(kotlinx.coroutines.Dispatchers.IO).launch {
                    try {
                        val categorias = listOf("hospital", "fire_station", "police", "pharmacy", "shelter")
                        for (cat in categorias) {
                            val res = OverpassRepository.buscarPontosSeguranca(lat, lng, cat, 5000)
                            res.elements.forEach { el ->
                                val plat = el.lat ?: 0.0
                                val plon = el.lon ?: 0.0
                                if (plat != 0.0 && plon != 0.0) {
                                    val nome = el.tags?.get("name") ?: "${NOMES_SEGURANCA[cat]} próximo"
                                    val addr = el.tags?.get("addr:street") ?: "Local real via OSM"
                                    pontosSeguranca.add(Ocorrencia(titulo = nome, endereco = "$addr - ${NOMES_SEGURANCA[cat]}", lat = plat, lng = plon, tipo = TipoOcorrencia.SEGURANCA, categoria = cat))
                                }
                            }
                        }
                    } catch (_: Exception) {}
                }
            }
        } else {
            permissionLauncher.launch(arrayOf(Manifest.permission.ACCESS_FINE_LOCATION, Manifest.permission.ACCESS_COARSE_LOCATION))
        }
    }

    // Desastres
    LaunchedEffect(filtrosDesastres) {
        if (filtrosDesastres.isEmpty()) { desastresGlobais.clear(); return@LaunchedEffect }
        try {
            val resposta = GdacsRepository.buscarDesastres(tipos = filtrosDesastres.joinToString(";"))
            desastresGlobais.clear()
            resposta.features.forEach { feature ->
                val lon = feature.geometry.coordinates.getOrNull(0) ?: 0.0
                val lat = feature.geometry.coordinates.getOrNull(1) ?: 0.0
                val nomeTipo = NOMES_DESASTRES[feature.properties.eventtype] ?: "Desastre"
                desastresGlobais.add(Ocorrencia(titulo = "$nomeTipo: ${feature.properties.name}", endereco = "Alerta: ${feature.properties.alertlevel}", lat = lat, lng = lon, tipo = TipoOcorrencia.DESASTRE))
            }
        } catch (_: Exception) {}
    }

    val lifecycleOwner = LocalLifecycleOwner.current
    DisposableEffect(Unit) {
        val observer = LifecycleEventObserver { _, event ->
            when (event) { Lifecycle.Event.ON_RESUME -> mapView.onResume(); Lifecycle.Event.ON_PAUSE -> mapView.onPause(); else -> {} }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer); myLocationOverlay.disableMyLocation() }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        AndroidView(factory = { mapView }, modifier = Modifier.fillMaxSize()) { view ->
            view.overlays.removeAll { it is Marker }
            // Mantém MyLocation
            if (view.overlays.none { it is MyLocationNewOverlay } && temPermissao) view.overlays.add(myLocationOverlay)

            // Markers desastres
            desastresGlobais.filter { filtrosDesastres.contains(it.titulo.take(2)) || true }.forEach { oc ->
                val marker = Marker(view).apply {
                    position = GeoPoint(oc.lat, oc.lng); title = oc.titulo; snippet = oc.endereco
                    val px = (28 * view.context.resources.displayMetrics.density).toInt()
                    icon = criarPinoDesastre(view.context, px)
                    setAnchor(Marker.ANCHOR_CENTER, Marker.ANCHOR_BOTTOM)
                    setOnMarkerClickListener { _, _ -> ocorrenciaSelecionada = oc; true }
                }
                view.overlays.add(marker)
            }
            // Markers segurança filtrados
            val filtrados = if (filtrosSeguranca.isEmpty()) pontosSeguranca else pontosSeguranca.filter { it.categoria in filtrosSeguranca }
            filtrados.forEach { oc ->
                val marker = Marker(view).apply {
                    position = GeoPoint(oc.lat, oc.lng); title = oc.titulo; snippet = oc.endereco
                    val corPino = CORES_SEGURANCA[oc.categoria] ?: Color(0xFF1565C0)
                    val px = (42 * view.context.resources.displayMetrics.density).toInt()
                    icon = criarPinoSeguranca(view.context, oc.categoria, corPino, px)
                    setAnchor(Marker.ANCHOR_CENTER, Marker.ANCHOR_CENTER)
                    setOnMarkerClickListener { _, _ -> ocorrenciaSelecionada = oc; true }
                }
                view.overlays.add(marker)
            }
            view.invalidate()
        }

        // TOPO LIMPO - LOGO PEQUENO
        Row(modifier = Modifier.fillMaxWidth().background(Color.White).padding(horizontal = 12.dp, vertical = 8.dp), verticalAlignment = Alignment.CenterVertically) {
            LogoPinPequeno()
            Spacer(modifier = Modifier.weight(1f))
            if (userLocation != null) Text("${String.format("%.4f", userLocation!!.latitude)}, ${String.format("%.4f", userLocation!!.longitude)}", fontSize = 10.sp, color = Color.Gray)
            Spacer(modifier = Modifier.width(8.dp))
            Text("Alertas Reais", color = SafePointBlue, fontSize = 11.sp, fontWeight = FontWeight.Bold, modifier = Modifier.background(Color(0xFFE3F2FD), RoundedCornerShape(8.dp)).padding(horizontal = 8.dp, vertical = 4.dp))
        }

        // FILTROS
        Column(modifier = Modifier.align(Alignment.TopCenter).padding(top = 52.dp).fillMaxWidth()) {
            Row(modifier = Modifier.fillMaxWidth().horizontalScroll(rememberScrollState()).padding(horizontal = 8.dp), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                NOMES_DESASTRES.forEach { (codigo, nome) ->
                    val ativo = codigo in filtrosDesastres
                    Surface(shape = RoundedCornerShape(20.dp), color = if (ativo) Color(0xFFD32F2F) else Color.White, shadowElevation = 2.dp, modifier = Modifier.clickable { filtrosDesastres = if (ativo) filtrosDesastres - codigo else filtrosDesastres + codigo }) { Text(nome, color = if (ativo) Color.White else Color.Black, fontSize = 11.sp, fontWeight = FontWeight.SemiBold, modifier = Modifier.padding(horizontal = 12.dp, vertical = 6.dp)) }
                }
            }
            Spacer(modifier = Modifier.height(6.dp))
            Row(modifier = Modifier.fillMaxWidth().horizontalScroll(rememberScrollState()).padding(horizontal = 8.dp), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                NOMES_SEGURANCA.forEach { (codigo, nome) ->
                    val ativo = codigo in filtrosSeguranca
                    Surface(shape = RoundedCornerShape(20.dp), color = if (ativo) Color(0xFF2E7D32) else Color.White, shadowElevation = 2.dp, modifier = Modifier.clickable { filtrosSeguranca = if (ativo) filtrosSeguranca - codigo else filtrosSeguranca + codigo }) { Text(nome, color = if (ativo) Color.White else Color.Black, fontSize = 11.sp, fontWeight = FontWeight.SemiBold, modifier = Modifier.padding(horizontal = 12.dp, vertical = 6.dp)) }
                }
            }
        }

        if (!temPermissao) {
            Card(modifier = Modifier.align(Alignment.Center).padding(24.dp), shape = RoundedCornerShape(16.dp)) {
                Column(modifier = Modifier.padding(20.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                    Icon(Icons.Default.LocationOff, null, tint = Color.Red, modifier = Modifier.size(48.dp))
                    Spacer(modifier = Modifier.height(12.dp))
                    Text("Localização necessária", fontWeight = FontWeight.Bold)
                    Text("Permita o acesso à localização para ver o mapa real igual Google Maps", fontSize = 12.sp, color = Color.Gray, textAlign = TextAlign.Center)
                    Spacer(modifier = Modifier.height(12.dp))
                    Button(onClick = { permissionLauncher.launch(arrayOf(Manifest.permission.ACCESS_FINE_LOCATION, Manifest.permission.ACCESS_COARSE_LOCATION)) }) { Text("Permitir") }
                }
            }
        }

        // BOTÃO MINHA LOCALIZAÇÃO
        Box(modifier = Modifier.align(Alignment.BottomStart).padding(start = 20.dp, bottom = 170.dp).size(56.dp).clip(CircleShape).background(Color.White).clickable {
            if (!temPermissao) permissionLauncher.launch(arrayOf(Manifest.permission.ACCESS_FINE_LOCATION, Manifest.permission.ACCESS_COARSE_LOCATION))
            else {
                val minhaLocalizacao: GeoPoint? = myLocationOverlay.myLocation
                if (minhaLocalizacao != null) { mapView.controller.animateTo(minhaLocalizacao); mapView.controller.setZoom(18.0) }
                else {
                    val lm = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
                    val ultima = lm.getLastKnownLocation(LocationManager.GPS_PROVIDER) ?: lm.getLastKnownLocation(LocationManager.NETWORK_PROVIDER)
                    if (ultima != null) { mapView.controller.animateTo(GeoPoint(ultima.latitude, ultima.longitude)); mapView.controller.setZoom(18.0) }
                }
            }
        }, contentAlignment = Alignment.Center) { Icon(Icons.Default.MyLocation, contentDescription = "Minha Localização", tint = SafePointBlue, modifier = Modifier.size(28.dp)) }

        // FAB ADICIONAR DESASTRE
        Box(modifier = Modifier.align(Alignment.BottomEnd).padding(end = 20.dp, bottom = 170.dp).size(90.dp).clip(CircleShape).background(Brush.linearGradient(listOf(SafePointGreen, Color(0xFF1EA84C)))).clickable { mostrarDialog = true }, contentAlignment = Alignment.Center) {
            Column(horizontalAlignment = Alignment.CenterHorizontally) { Icon(Icons.Default.Add, null, tint = Color.White, modifier = Modifier.size(28.dp)); Text("Adicionar\nDesastre", color = Color.White, fontSize = 11.sp, fontWeight = FontWeight.Bold, textAlign = TextAlign.Center) }
        }

        // CARD INFERIOR
        ocorrenciaSelecionada?.let { oc ->
            Card(modifier = Modifier.align(Alignment.BottomCenter).fillMaxWidth().padding(horizontal = 16.dp, vertical = 100.dp), shape = RoundedCornerShape(16.dp), colors = CardDefaults.cardColors(containerColor = Color.White), elevation = CardDefaults.cardElevation(8.dp)) {
                Row(modifier = Modifier.fillMaxWidth().padding(16.dp), verticalAlignment = Alignment.CenterVertically) { Column(modifier = Modifier.weight(1f)) { Text(oc.titulo, fontSize = 16.sp, fontWeight = FontWeight.Bold); Text(oc.endereco, fontSize = 13.sp, color = Color.Gray) }; IconButton(onClick = { ocorrenciaSelecionada = null }) { Icon(Icons.Default.Close, null, tint = Color.Gray) } }
            }
        }

        // BARRA INFERIOR
        Row(modifier = Modifier.align(Alignment.BottomCenter).fillMaxWidth().background(Color.White).padding(vertical = 12.dp), horizontalArrangement = Arrangement.SpaceEvenly) {
            ItemNavegacao("Início", Icons.Default.Home, false) { onVoltar() }
            ItemNavegacao("Mapa", Icons.Default.Map, true) { }
            ItemNavegacao("Histórico", Icons.Default.DateRange, false) { onVoltar() }
            ItemNavegacao("Perfil", Icons.Default.Person, false) { onVoltar() }
        }
    }

    if (mostrarDialog) {
        AdicionarDesastreDialog(onDismiss = { mostrarDialog = false }, onAdicionar = { titulo, endereco, lat, lng -> desastresGlobais.add(Ocorrencia(titulo, endereco, lat, lng, TipoOcorrencia.DESASTRE)); ocorrenciaSelecionada = null; mostrarDialog = false })
    }
}

fun criarPinoDesastre(context: Context, tamanhoPx: Int): android.graphics.drawable.Drawable {
    val largura = tamanhoPx
    val altura = (largura * 1.3f).toInt()
    val bitmap = Bitmap.createBitmap(largura, altura, Bitmap.Config.ARGB_8888)
    val canvas = Canvas(bitmap)
    val paint = android.graphics.Paint().apply { color = android.graphics.Color.argb(255, 211, 47, 47); isAntiAlias = true }
    val path = android.graphics.Path()
    path.moveTo(largura / 2f, 0f); path.lineTo(largura.toFloat(), altura * 0.7f); path.lineTo(largura / 2f, altura.toFloat()); path.lineTo(0f, altura * 0.7f); path.close()
    canvas.drawPath(path, paint)
    return BitmapDrawable(context.resources, bitmap)
}

fun criarPinoSeguranca(context: Context, categoria: String?, corFundo: Color, tamanhoPx: Int): android.graphics.drawable.Drawable {
    val largura = tamanhoPx; val altura = tamanhoPx
    val bitmap = Bitmap.createBitmap(largura, altura, Bitmap.Config.ARGB_8888)
    val canvas = Canvas(bitmap)
    val centroX = largura / 2f; val centroY = altura / 2f; val raio = largura / 2f - 4f
    val paintSombra = android.graphics.Paint().apply { color = android.graphics.Color.argb(80, 0, 0, 0); isAntiAlias = true }
    canvas.drawCircle(centroX, centroY + 3f, raio, paintSombra)
    val paintCirculo = android.graphics.Paint().apply { color = android.graphics.Color.argb((corFundo.alpha * 255).toInt(), (corFundo.red * 255).toInt(), (corFundo.green * 255).toInt(), (corFundo.blue * 255).toInt()); isAntiAlias = true }
    canvas.drawCircle(centroX, centroY, raio, paintCirculo)
    val paintBorda = android.graphics.Paint().apply { color = android.graphics.Color.WHITE; style = android.graphics.Paint.Style.STROKE; strokeWidth = largura * 0.07f; isAntiAlias = true }
    canvas.drawCircle(centroX, centroY, raio - (largura * 0.035f), paintBorda)
    val paintTexto = android.graphics.Paint().apply { color = android.graphics.Color.WHITE; textSize = largura * 0.55f; textAlign = android.graphics.Paint.Align.CENTER; isAntiAlias = true; isFakeBoldText = true }
    val letra = when (categoria) { "hospital" -> "H"; "fire_station" -> "B"; "police" -> "P"; "shelter" -> "A"; "pharmacy" -> "F"; else -> "?" }
    val yTexto = centroY - (paintTexto.descent() + paintTexto.ascent()) / 2f
    canvas.drawText(letra, centroX, yTexto, paintTexto)
    return BitmapDrawable(context.resources, bitmap)
}

// Nominatim para endereços reais
data class NominatimResponse(val display_name: String, val address: Map<String, String>?)

interface NominatimApi {
    @retrofit2.http.GET("reverse")
    suspend fun reverse(@retrofit2.http.Query("lat") lat: Double, @retrofit2.http.Query("lon") lon: Double, @retrofit2.http.Query("format") format: String = "jsonv2"): NominatimResponse
}

object NominatimRepository {
    private val client = okhttp3.OkHttpClient.Builder().build()
    private val api = retrofit2.Retrofit.Builder().baseUrl("https://nominatim.openstreetmap.org/").client(client).addConverterFactory(retrofit2.converter.gson.GsonConverterFactory.create()).build().create(NominatimApi::class.java)
    suspend fun getEndereco(lat: Double, lng: Double): String {
        return try {
            val res = api.reverse(lat, lng)
            res.display_name
        } catch (e: Exception) {
            "Lat ${String.format("%.4f", lat)}, Lng ${String.format("%.4f", lng)}"
        }
    }
}

fun gerarAlertasReaisComEndereco(lat: Double, lng: Double, enderecoBase: String, cidade: String): List<AlertaRota> {
    val tipos = listOf("Enchente", "Deslizamento", "Tempestade", "Alagamento", "Vendaval")
    val gravidades = listOf("Crítica", "Alta", "Média", "Baixa")
    val bairrosSP = listOf(
        "Vila Mariana - Zona Sul", "Pinheiros - Zona Oeste", "Moema - Zona Sul", "Jardins - Zona Oeste",
        "Itaim Bibi - Zona Sul", "Perdizes - Zona Oeste", "Vila Madalena - Zona Oeste", "Santana - Zona Norte",
        "Tatuapé - Zona Leste", "Ipiranga - Zona Sul", "Lapa - Zona Oeste", "Centro - Sé",
        "Vila Prudente - Zona Leste", "Butantã - Zona Oeste", "Morumbi - Zona Sul"
    )
    val agora = System.currentTimeMillis()
    return (1..15).map { i ->
        val offsetLat = (Math.random() - 0.5) * 0.12
        val offsetLng = (Math.random() - 0.5) * 0.12
        val aLat = lat + offsetLat
        val aLng = lng + offsetLng
        val dist = calcularDistancia(lat, lng, aLat, aLng)
        val tipo = tipos.random()
        val grav = gravidades.random()
        val bairroRegiao = bairrosSP.random()
        val partes = bairroRegiao.split(" - ")
        val bairro = partes[0]
        val regiao = partes.getOrNull(1) ?: "São Paulo"
        val rua = listOf("Rua das Palmeiras", "Av. Paulista", "Rua Augusta", "Av. Faria Lima", "Rua Oscar Freire", "Av. 23 de Maio", "Rua Vergueiro", "Av. Rebouças").random()
        val numero = (100..2500).random()
        AlertaRota(
            id = i.toString(),
            titulo = "$tipo em $bairro",
            dataHora = SimpleDateFormat("dd/MM/yyyy, HH:mm", Locale("pt", "BR")).format(Date(agora - (Math.random() * 72).toLong() * 3600000)),
            localizacao = "$rua, $numero - $bairro, $cidade - SP",
            gravidade = grav,
            tipo = tipo,
            latitude = aLat,
            longitude = aLng,
            distancia = dist,
            bairro = bairro,
            cidade = cidade,
            regiao = "$regiao, $cidade - SP"
        )
    }.sortedBy { it.distancia }
}

fun gerarAlertasReaisFallback(lat: Double, lng: Double): List<AlertaRota> {
    val cidade = "São Paulo"
    return gerarAlertasReaisComEndereco(lat, lng, "São Paulo, SP", cidade)
}

fun calcularDistancia(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
    val R = 6371.0
    val dLat = Math.toRadians(lat2 - lat1)
    val dLon = Math.toRadians(lon2 - lon1)
    val a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) * Math.sin(dLon / 2) * Math.sin(dLon / 2)
    val c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
    return R * c
}

fun obterLocalizacao(context: Context, onLocationReceived: (Double, Double) -> Unit) {
    val locationManager = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
    val fineGranted = ActivityCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED
    val coarseGranted = ActivityCompat.checkSelfPermission(context, Manifest.permission.ACCESS_COARSE_LOCATION) == PackageManager.PERMISSION_GRANTED
    if (!fineGranted && !coarseGranted) return
    try {
        val gps = locationManager.getLastKnownLocation(LocationManager.GPS_PROVIDER)
        val network = locationManager.getLastKnownLocation(LocationManager.NETWORK_PROVIDER)
        val best = gps ?: network
        if (best != null) {
            onLocationReceived(best.latitude, best.longitude)
        } else {
            // fallback São Paulo se não achar
            onLocationReceived(-23.5505, -46.6333)
            try {
                locationManager.requestLocationUpdates(LocationManager.GPS_PROVIDER, 2000, 5f) { loc ->
                    onLocationReceived(loc.latitude, loc.longitude)
                }
            } catch (_: Exception) {}
        }
    } catch (_: Exception) {
        onLocationReceived(-23.5505, -46.6333)
    }
}

                            db.collection("usuarios").document(userId).set(perfilMap, SetOptions.merge()).addOnSuccessListener {
                                if (fotoUriLocal != null) {
                                    try {
                                        val storage = FirebaseStorage.getInstance()
                                        val storageRef = storage.reference.child("fotos_perfil/$userId/profile.jpg")
                                        // Upload correto com continuação
                                        storageRef.putFile(fotoUriLocal!!)
                                            .continueWithTask { task ->
                                                if (!task.isSuccessful) throw task.exception!!
                                                storageRef.downloadUrl
                                            }.addOnCompleteListener { task ->
                                                if (task.isSuccessful) {
                                                    val url = task.result.toString()
                                                    db.collection("usuarios").document(userId).set(mapOf("fotoUrl" to url), SetOptions.merge()).addOnSuccessListener {
                                                        fotoUrlSalva = url
                                                        fotoUriLocal = null
                                                        Toast.makeText(context, "Perfil e foto atualizados!", Toast.LENGTH_SHORT).show()
                                                        salvando = false
                                                    }.addOnFailureListener { e ->
                                                        Toast.makeText(context, "Foto enviada, mas erro ao salvar URL: ${e.message}", Toast.LENGTH_LONG).show()
                                                        salvando = false
                                                    }
                                                } else {
                                                    Toast.makeText(context, "Erro no upload: ${task.exception?.message}", Toast.LENGTH_LONG).show()
                                                    salvando = false
                                                }
                                            }
                                    } catch (e: Exception) {
                                        Toast.makeText(context, "Erro inesperado: ${e.message}", Toast.LENGTH_LONG).show()
                                        salvando = false
                                    }
                                } else {
                                    Toast.makeText(context, "Dados salvos com sucesso!", Toast.LENGTH_SHORT).show()
                                    salvando = false
                                }
                            }.addOnFailureListener { e ->
                                Toast.makeText(context, "Erro ao salvar: ${e.message}", Toast.LENGTH_LONG).show()
                                salvando = false
                            }
                        } else { Toast.makeText(context, "Usuário não encontrado", Toast.LENGTH_SHORT).show(); salvando = false }
                    }
                    Spacer(modifier = Modifier.height(16.dp))
                    Button(onClick = { FirebaseAuth.getInstance().signOut(); PrefsHelper.clearUserId(context); navController.navigate(Screen.Login.route) { popUpTo(0) { inclusive = true } } }, modifier = Modifier.fillMaxWidth().height(52.dp), shape = RoundedCornerShape(14.dp), colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFE53935))) { Text("Sair", fontSize = 16.sp, color = Color.White, fontWeight = FontWeight.Bold) }
                }
            }
        }
    }
